<?php
require_once 'db.php';
if (!isset($_SESSION['user_id'])) { header("Location: login.php"); exit(); }

$user_id = $_SESSION['user_id'];
$user = $conn->query("SELECT * FROM users WHERE id=$user_id")->fetch_assoc();

// ইনভেস্ট প্ল্যান কেনা
if (isset($_POST['invest'])) {
    $amount = $_POST['amount'];
    $daily_profit = $amount * 0.05; // দৈনিক ৫% লাভ

    if ($user['balance'] >= $amount) {
        $conn->query("UPDATE users SET balance = balance - $amount WHERE id=$user_id");
        $conn->query("INSERT INTO investments (user_id, amount, daily_profit) VALUES ($user_id, $amount, $daily_profit)");
        $msg = "Investment Successful!";
        header("Refresh:0");
    } else {
        $error = "Insufficient Balance!";
    }
}

// ডিপোজিট রিকোয়েস্ট
if (isset($_POST['deposit'])) {
    $amount = $_POST['amount'];
    $method = $_POST['method'];
    $trx_id = $_POST['trx_id'];

    $conn->query("INSERT INTO deposits (user_id, amount, method, trx_id) VALUES ($user_id, $amount, '$method', '$trx_id')");
    $msg = "Deposit request submitted!";
}
